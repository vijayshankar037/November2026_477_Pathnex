# Ansible - Multi-Node Nginx setup

- name: Setup Nginx on multiple nodes
  hosts: all
  become: yes

  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present

    - name: Start Nginx service
      service:
        name: nginx
        state: started
        enabled: yes

## Kubernates - Deploy Nginx with ConfigMap and Secrets

---
apiVersion: v1
kind: ConfigMap
metadata:
  name: nginx-config
data:
  nginx.conf: |
    server {
      listen 80;
      server_name localhost;

      location / {
        root /usr/share/nginx/html;
        index index.html index.htm;
      }
    }

---
apiVersion: v1
kind: Secret
metadata:
  name: nginx-secret
type: Opaque
data:
  password: cGF0aG5leHBhc3M=

---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-nginx
spec:
  replicas: 2

  selector:
    matchLabels:
      app: pathnex-nginx

  template:
    metadata:
      labels:
        app: pathnex-nginx

    spec:
      containers:
        - name: nginx
          image: nginx:latest
          ports:
            - containerPort: 80

          volumeMounts:
            - name: nginx-config
              mountPath: /etc/nginx/nginx.conf
              subPath: nginx.conf
              readOnly: true

            - name: nginx-secret
              mountPath: /etc/nginx/secrets
              readOnly: true

      volumes:
        - name: nginx-config
          configMap:
            name: nginx-config

        - name: nginx-secret
          secret:
            secretName: nginx-secret


## Jenkinsfile- Multi-Branch Pipeline with Docker Build

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                script {
                    docker.build('pathnex-app')
                }
            }
        }

        stage('Deploy') {
            steps {
                script {
                    sh 'kubectl apply -f kubernetes/deployment.yaml'
                }
            }
        }
    }
}


## GitLab CI/CD - Advance Deployment with Helm

stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - helm repo add pathnex https://charts.pathnex.com
    - helm repo update
    - helm install pathnex-nginx pathnex/nginx-ingress


## Docker
# COPY vs CMD

FROM ubuntu:latest

# COPY: Copies files from the build context
COPY pathnex-file.txt /opt/pathnex/files/

# ADD: Copies files and automatically extracts local tar archives
ADD pathnex.tar.gz /opt/pathnex/extracted/
