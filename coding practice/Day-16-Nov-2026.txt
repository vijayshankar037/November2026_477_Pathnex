## Kubernates - Multi-Cluster Deployment with Federation

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
  labels:
    app: pathnex-app

spec:
  replicas: 2

  selector:
    matchLabels:
      app: pathnex-app

  template:
    metadata:
      labels:
        app: pathnex-app

    spec:
      containers:
        - name: nginx
          image: nginx:latest
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: pathnex-service

spec:
  selector:
    app: pathnex-app

  ports:
    - protocol: TCP
      port: 80
      targetPort: 80

  type: ClusterIP


## Jenkinsfile - Deploy to Multi-Cluster Environments

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('pathnex-web-app')
                }
            }
        }

        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.CLUSTER == 'us-east-1') {
                        sh '''
                            kubectl --context=us-east \
                            apply -f kubernetes/deployment.yaml
                        '''
                    } else if (env.CLUSTER == 'us-west-2') {
                        sh '''
                            kubectl --context=us-west \
                            apply -f kubernetes/deployment.yaml
                        '''
                    } else {
                        error "Unsupported cluster: ${env.CLUSTER}"
                    }
                }
            }
        }
    }
}


## GitLab CI/CD - Mutli-Region Deployment with Kubernates

stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE/pathnex-web-app"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_NAME:$CI_COMMIT_SHA" .

push:
  stage: push
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker push "$IMAGE_NAME:$CI_COMMIT_SHA"

deploy:
  stage: deploy
  script:
    - |
      if [ "$DEPLOY_REGION" = "us-east-1" ]; then
        kubectl --context=us-east apply -f kubernetes/prod-deployment.yaml
      elif [ "$DEPLOY_REGION" = "us-west-2" ]; then
        kubectl --context=us-west apply -f kubernetes/dev-deployment.yaml
      else
        echo "Invalid deployment region"
        exit 1
      fi
